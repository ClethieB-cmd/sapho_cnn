1 // CNN completa num unico processador SAPHO. Conv1(32,k5)->ReLU->BN1 ->
2 // Conv2(16,k3)->ReLU->BN2 -> Dense(64)->ReLU -> Dense(32)->ReLU -> Dense(1).
3 // Pesos = CNN TREINADA. BN explicita (a*x+b). I/O inteiro: entrada = int(amostra*2^20),
4 // descalada dentro; saida = int(pred*2^20).
5 #NUBITS 32
6 #NBMANT 23
7 #NBEXPO 8
8 #NDSTAC 64
9 #SDEPTH 16
10 #NUIOIN 1
11 #NUIOOU 1
12 #NUGAIN 128
13 #FROUND 2

15 #define SCALE_IN  1048576
16 #define SCALE_OUT 1048576
17 
18 float conv1_W[160]  "conv1_W.txt";
19 float conv1_b[32]   "conv1_b.txt";
20 float bn1_a[32]     "bn1_a.txt";
21 float bn1_b[32]     "bn1_b.txt";
22 float conv2_W[1536] "conv2_W.txt";
23 float conv2_b[16]   "conv2_b.txt";
24 float bn2_a[16]     "bn2_a.txt";
25 float bn2_b[16]     "bn2_b.txt";
26 float dense1_W[7168] "dense1_W.txt";
27 float dense1_b[64]   "dense1_b.txt";
28 float dense2_W[2048] "dense2_W.txt";
29 float dense2_b[32]   "dense2_b.txt";
30 float dense3_W[32]   "dense3_W.txt";
31 float dense3_b[1]    "dense3_b.txt";
32 
33 float in_buf[7];
34 float buf_conv1[7][32];
35 float buf_conv2[7][16];
36 float buf_flat[112];
37 float buf_d1[64];
38 float buf_d2[32];
39 
40 int pos; int fx; int kx; int cx; int src; int ii; int jx;
41 float acc; float tmp; float pred;
42 
43 void conv1_layer() {
44     pos = 0;
45     while (pos < 7) {
46         fx = 0;
47         while (fx < 32) {
48             acc = conv1_b[fx];
49             kx = 0;
50             while (kx < 5) {
51                 src = pos + kx - 2;
52                 if (src >= 0 && src < 7) { tmp = in_buf[src] * conv1_W[fx*5 + kx]; acc = acc + tmp; }
53                 kx = kx + 1;
54             }
55             acc = pset(acc);
56             buf_conv1[pos][fx] = acc*bn1_a[fx] + bn1_b[fx];
57             fx = fx + 1;
58         }
59         pos = pos + 1;
60     }
61 }
62 
63 void conv2_layer() {
64     pos = 0;
65     while (pos < 7) {
66         fx = 0;
67         while (fx < 16) {
68             acc = conv2_b[fx];
69             kx = 0;
70             while (kx < 3) {
71                 src = pos + kx - 1;
72                 if (src >= 0 && src < 7) {
73                     cx = 0;
74                     while (cx < 32) { tmp = buf_conv1[src][cx] * conv2_W[(fx*3 + kx)*32 + cx]; acc = acc + tmp; cx = cx + 1; }
75                 }
76                 kx = kx + 1;
77             }
78             acc = pset(acc);
79             buf_conv2[pos][fx] = acc*bn2_a[fx] + bn2_b[fx];
80             fx = fx + 1;
81         }
82         pos = pos + 1;
83     }
84 }
85 
86 void dense_layers() {
87     pos = 0;
88     while (pos < 7) { jx = 0; while (jx < 16) { buf_flat[pos*16 + jx] = buf_conv2[pos][jx]; jx = jx + 1; } pos = pos + 1; }
89     jx = 0;
90     while (jx < 64) { acc = dense1_b[jx]; ii = 0; while (ii < 112) { tmp = buf_flat[ii] * dense1_W[ii*64 + jx]; acc = acc + tmp; ii = ii + 1; } buf_d1[jx] = pset(acc); jx = jx + 1; }
91     jx = 0;
92     while (jx < 32) { acc = dense2_b[jx]; ii = 0; while (ii < 64) { tmp = buf_d1[ii] * dense2_W[ii*32 + jx]; acc = acc + tmp; ii = ii + 1; } buf_d2[jx] = pset(acc); jx = jx + 1; }
93     acc = dense3_b[0]; ii = 0; while (ii < 32) { tmp = buf_d2[ii] * dense3_W[ii]; acc = acc + tmp; ii = ii + 1; } pred = acc;
94 }
95 
96 void main() {
97     while (1) {
98         ii = 0; while (ii < 7) { in_buf[ii] = fin(0) / SCALE_IN; ii = ii + 1; }
99         conv1_layer(); conv2_layer(); dense_layers();
100         fout(0, pred * SCALE_OUT);
101     }
102 }
